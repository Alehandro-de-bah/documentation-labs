```cpp
#include <iostream>
#include <cmath>

bool isPrime(int number) {
    if (number <= 1) {
        return false;  // числа ≤ 1 не являются простыми
    }
    
    if (number == 2) {
        return true;   // 2 - единственное четное простое число
    }
    
    if (number % 2 == 0) {
        return false;  // все остальные четные числа - составные
    }
    
    // 2. Проверяем нечетные делители до квадратного корня из числа
    // Если number = a * b, то один из множителей ≤ sqrt(number)
    int limit = sqrt(number);
    
    for (int i = 3; i <= limit; i += 2) {
        // i += 2 - проверяем только нечетные числа
        if (number % i == 0) {
            return false;  // нашелся делитель - число составное
        }
    }
    
    return true;  // делителей не найдено - число простое
}
```
